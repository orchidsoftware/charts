Choose the visual grammar. The same lifecycle handles responsive rendering, updates, interaction, accessibility, export, and cleanup. 

import { createChart } from "@charts2/core";
import "@charts2/core/style.css";

const chart = createChart("#revenue", {
  type: "bar",
  orientation: "horizontal",
  data: { datasets: [{ values: [18, 31, 44] }] }
});

chart.update(nextData);
chart.destroy();
