The code behind this chart

Add <div id="your-chart"></div> to your page, then:

import { LineChart } from "@orchidsoftware/charts";
import "@orchidsoftware/charts/style.css";

const chart = LineChart.make("#your-chart")
  .labels(["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul"])
  .dataset("Revenue", [4, 8, 15, 16, 23, 42, 108])
  .colors(["#8267aa"])
  .height(260)
  .render();
